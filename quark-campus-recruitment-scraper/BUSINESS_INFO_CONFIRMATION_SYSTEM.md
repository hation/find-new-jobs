#!/usr/bin/env python3
"""
🤖 夸克项目自动确认系统
版本: 1.0
功能: 检测到关键词后请求用户确认业务信息
"""

class BusinessInfoConfirmation:
    def __init__(self):
        self.detected_keywords = []
        self.required_info = {
            "list_url": None,
            "target_positions": None,
            "filter_categories": None,
            "core_fields": None,
            "default_strategy": None
        }
    
    def detect_and_request_confirmation(self, user_message):
        """检测关键词并请求确认"""
        keywords = self._detect_keywords(user_message)
        
        if not keywords:
            return None
            
        confirmation_request = self._build_confirmation_request(keywords)
        return confirmation_request
    
    def _detect_keywords(self, message):
        """检测关键词"""
        detected = []
        
        # 检测任何包含"夸克"的消息
        if "夸克" in message:
            detected.append("夸克相关任务")
            
        # 检测招聘相关关键词
        recruitment_keywords = ["招聘", "爬取", "数据提取", "岗位"]
        for keyword in recruitment_keywords:
            if keyword in message:
                detected.append(keyword)
                
        return detected
    
    def _build_confirmation_request(self, keywords):
        """构建确认请求"""
        return f"""
🎯 **检测到关键词: {', '.join(keywords)}**

请提供或确认以下信息：

### 2. 🎯 **业务目标**
- 目标岗位数: [例如: 92个]

### 3. 📊 **数据规范**

## 🎯 **为什么需要这个系统？**

### **AI的限制**
1. ❌ **无法记忆**: 新会话中忘记所有信息
2. ❌ **可能假设错误**: 如我假设了错误的网址
3. ❌ **信息可能过时**: 业务需求可能变化

### **系统的优势**
1. ✅ **信息准确**: 用户提供，确保正确
2. ✅ **灵活适应**: 可以处理不同项目
3. ✅ **可验证**: 信息明确，可验证
4. ✅ **用户控制**: 用户完全控制业务信息
