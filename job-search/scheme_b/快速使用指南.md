### **问题3：数据不匹配**
```bash
# 验证职位ID匹配
python3 -c "
import json
import pandas as pd

# 加载Excel数据
df = pd.read_excel('深圳AI岗位_职位ID合并_*.xlsx')
excel_ids = set(df['职位ID'].dropna().astype(str))

# 加载详情数据
import glob
detail_files = glob.glob('职位详情/detail_*.json')
detail_ids = set()

for file in detail_files[:3]:
    try:
        with open(file, 'r') as f:
            data = json.load(f)
        job_id = data.get('data', {}).get('jobInfo', {}).get('encryptId', '')
        if job_id:
            detail_ids.add(job_id)
    except:
        continue

print(f'Excel中的职位ID: {len(excel_ids)}个')
print(f'详情中的职位ID: {len(detail_ids)}个')
print(f'匹配数量: {len(excel_ids & detail_ids)}个')
"
```
