### **🎯 六大核心原则**

| 原则 | 核心思想 | 关键实践 | 适用场景 |
|------|----------|----------|----------|
| **1. 数据格式适应性** | 外部工具可能返回不同格式 | 多格式解析，保持原始数据 | API/工具调用 |
| **2. 数据清洗前置** | 早期清洗成本最低 | 输入验证，格式标准化 | 数据导入/处理 |
| **3. 错误诊断分层** | 表面错误可能掩盖根本问题 | 错误分类，分层处理 | 错误调试 |
| **4. 环境配置显式化** | 隐式依赖导致运行时失败 | 显式PATH设置，依赖检查 | 环境配置 |
| **5. 字段映射验证** | 永远基于实际数据验证字段名 | 实际数据检查，映射表 | 数据合并 |
| **6. 保守执行策略** | 在不确定环境中保守更可靠 | 连续失败停止，智能跳过 | 资源受限操作 |

```python
def diagnose_error(error_data):
    """分层诊断错误"""
    error_code = error_data.get('code', '')
    error_msg = error_data.get('message', '')
    
    # 第一层：认证错误（根本错误）
    if error_code == 'not_authenticated':
        return {
            'level': 'critical',
            'type': 'authentication',
            'message': f'Token已过期: {error_msg}',
            'action': '重新登录: boss logout && boss login'
        }
    
    # 第二层：数据错误（表面错误）
    elif error_code == 200301:
        return {
            'level': 'warning',
            'type': 'data_invalid',
            'message': f'职位不存在: {error_msg}',
            'action': '跳过此security_id，继续下一个'
        }
    
    # 第三层：其他错误
    else:
        return {
            'level': 'error',
            'type': 'unknown',
            'message': f'未知错误: {error_code} - {error_msg}',
```

```python
def conservative_fetch_strategy(security_ids, max_attempts=10, max_failures=2):
    """保守获取策略"""
    success_count = 0
    failure_count = 0
    consecutive_failures = 0
    should_stop = False
    
    for i, security_id in enumerate(security_ids[:max_attempts]):
        if should_stop:
            print("🛑 达到停止条件，提前结束")
            break
        
        print(f"📋 尝试第 {i+1} 个security_id")
        
        try:
            result = fetch_detail(security_id)
            if result['success']:
                success_count += 1
                consecutive_failures = 0  # 成功时重置失败计数
                print(f"✅ 获取成功 ({success_count}/{i+1})")
            else:
                failure_count += 1
                consecutive_failures += 1
                
                # 检查是否应该停止
                if consecutive_failures >= max_failures:
                    print(f"💡 连续失败{consecutive_failures}次，可能有问题，停止获取")
                    should_stop = True
                
                print(f"❌ 获取失败 ({failure_count}/{i+1})")
        
        except Exception as e:
            failure_count += 1
            consecutive_failures += 1
            print(f"⚠️  异常失败: {e}")
    
    return {
        'attempted': min(max_attempts, len(security_ids)),
        'success': success_count,
        'failure': failure_count,
        'stopped_early': should_stop
    }
```
